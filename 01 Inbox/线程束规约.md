利用 CUDA 内置的函数，在一个线程束内进行高效规约，具体代码如下：

```cpp
template<const int kWarpSize=WARP_SIZE>
__device__ __forceinline__ float warp_reduce_sum(float val) {
#pragma unroll
	for (int mask = WARP_SIZE >> 1; mask >= 1; mask >>= 1) {
		val += __shfl_xor_sync(0xffffffff, val, mask);
	}
}

template<const int NUM_THREADS = 256>
__device__ void block_reduce_sum_f32(float val) {
	constexpr int NUM_WARPS = (NUM_THREADS + WARP_SIZE - 1) / WARP_SIZE;
	int warp = threadIdx.x / WARP_SIZE;
	int lane = threadIdx.x % WARP_SIZE;
	
	static __shared__ float shared[NUM_WARPS];
	val = warp_reduce_sum<WARP_SIZE>(val);

	if (lane == 0) shared[warp] = val;
	__syncthreads();
	val = (lane < NUM_WARPS) ? shard[lane] : 0.0f;
	val = warp_reduce_sum<NUM_WAPS>(val);
	return val;
}
```

注意细节：
- 对于一个 Block 而言，它内部的线程束数量一定是小于 32 的，因为每个 Block 的线程数上限是 1024。所以可以对 shared 进行线程束规约；
- **__shfl_xor_sync(mask=0xffffffff,  val=val,  offset=mask)** 这个函数:
  Mask：表示需要执行的线程 id；
  Val：表示需要交换的寄存器值；
  Offset：表示交换的偏移量；
